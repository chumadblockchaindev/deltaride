import { useState, useEffect } from 'react';
import { motion } from 'framer-motion';
import api from '../../api/api';
import { CarInterface } from './CarBooking';

interface Booking {
  car_id: number;
  start_date: string;
  end_date: string;
  created_at: string;
  id: number;
  car_detail?: CarInterface;
}

function BookingHistory() {
  const [bookings, setBookings] = useState<Booking[]>([]);

  // Fetch bookings once
  useEffect(() => {
    async function fetchBooking() {
      try {
        const res = await api.get('/api/booking/history/');
        setBookings(res.data);
      } catch (err) {
        console.error(err);
      }
    }
    fetchBooking();
  }, []);

  // Fetch car details once after bookings are loaded
  useEffect(() => {
    async function getCarDetails(data: Booking[]) {
      for (let i = 0; i < data.length; i++) {
        try {
          const res = await api.post('/api/cars/detail/', { car_id: data[i].car_id });
          setBookings((prevData) =>
            prevData.map((item) =>
              item.car_id === res.data.id
                ? {
                    ...item,
                    car_detail: res.data,
                  }
                : item
            )
          );
        } catch (err) {
          console.error("Error fetching car data", err);
        }
      }
    }

    if (bookings.length > 0) {
      getCarDetails(bookings);
    }
  }, [bookings.length]); // only run once after bookings are initially set

  return (
    <motion.div
      className="p-8 max-w-4xl mx-auto"
      initial={{ y: 30, opacity: 0 }}
      animate={{ y: 0, opacity: 1 }}
      transition={{ duration: 0.6 }}
    >
      <h1 className="text-4xl font-bold mb-8 text-center">📜 Booking History</h1>
      {bookings.length === 0 ? (
        <p className="text-center">No bookings yet.</p>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          {bookings.map((booking: Booking) => (
            <div key={booking.id} className="border p-4 rounded-xl shadow hover:shadow-lg transition flex justify-between">
              <div>
                <div className='flex gap-3'>
                  <h3 className="font-semibold">{booking.car_detail?.model_name || 'Loading car...'}</h3>
                  <h3 className="font-semibold">{booking.car_detail?.brand.name || 'Loading car...'}</h3>
                </div>
                <p>Date Booked: {new Date(booking.created_at).toLocaleDateString()}</p>
                <h3 className="font-semibold">
                  Booking: {booking.start_date} → {booking.end_date}
                </h3>
              </div>

              <img src={booking.car_detail?.image}  alt={booking.car_detail?.brand.name} className="w-15 h-15 object-cover rounded" />
            </div>
          ))}
        </div>
      )}
    </motion.div>
  );
}

export default BookingHistory;
